\bigskip
Offset Syntax (Offset)
\bi
      \ite \# expr
      \ite \{-\}Rn\\
      where\\
      \begin{tabular}{l l}
         expr & integer value between -255 and +255\\
	    & \\
	 - & negative result of the address calculation\\
	 Rn & address register\\
      \end{tabular}

\ei

\bigskip
Flexible Offset Syntax (FlexOffset)
\bi
      \ite \# expr
      \ite \{-\}Rn\{shift\}\\
      where\\
      \begin{tabular}{l l}
         expr & integer value between -4095 and +4095\\
	    & \\
	 - & negative result of the address calculation\\
	 Rn & address register\\
	 shift & ASR n  arithmetic shift right n bits 1 \underline{\textless} n \underline{\textless} 32\\
	       & LSL n 1 logical shift left n bits 0 \underline{\textless} n \underline{\textless} 31\\
	       & LSR n 1 logical shift right n bits 1 \underline{\textless} n \underline{\textless} 32\\
	       & ROR n 1 rotate right n bits 1 \underline{\textless} n \underline{\textless} 31\\
	       & RRX n  rotate right one bit, with extend\\
      \end{tabular}

\ei
